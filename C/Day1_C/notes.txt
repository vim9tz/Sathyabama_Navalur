 // Stage 1: Preprocessing (-E)
gcc -E hello.c -o hello.i
 // Stage 2: Compilation (-S)
gcc -S hello.c -o hello.s
 // Stage 3: Assembly (-C)
gcc -c hello.c -o hello.o
// Stage 4: Execution (Final Executable)
gcc hello.o -o hello
