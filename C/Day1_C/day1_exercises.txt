PROBLEM 1: Coordinate Swap Using Temporary Variable
--------------------------------------------------------------------------------
[Problem Description]
Swaps two user-provided coordinates using a temporary holding variable 
(Profile 1) and displays the values before and after swapping.

Sample Input:
10
20

Sample Output:
Profiles before swap is 10 and 20.
Profiles after swap is 20 and 10.

[Solution Code]
#include <stdio.h>

int main() {
    int p1, p2, temp;
    if (scanf("%d %d", &p1, &p2) != 2) return 1;

    printf("Profiles before swap is %d and %d.\n", p1, p2);

    temp = p1;
    p1 = p2;
    p2 = temp;

    printf("Profiles after swap is %d and %d.\n", p1, p2);
    return 0;
}

PROBLEM 2: Coordinate Swap Without Temporary Variable
--------------------------------------------------------------------------------
[Problem Description]
Swaps two user-provided coordinates using pure math without a third variable 
(Profile 2) and displays the values before and after swapping.

Sample Input:
10
20

Sample Output:
Profiles before swap is 10 and 20.
Profiles after swap is 20 and 10.

[Solution Code]
#include <stdio.h>

int main() {
    int p1, p2;
    if (scanf("%d %d", &p1, &p2) != 2) return 1;

    printf("Profiles before swap is %d and %d.\n", p1, p2);

    p1 = p1 + p2;
    p2 = p1 - p2;
    p1 = p1 - p2;

    printf("Profiles after swap is %d and %d.\n", p1, p2);
    return 0;
}

PROBLEM 3: Mixed-Type Integer and Float Operations
--------------------------------------------------------------------------------
[Problem Description]
Reads two integers on Line 1 and two floats on Line 2. Outputs the integer sum 
and difference, followed by the float sum and difference rounded to 1 decimal place.

Sample Input:
10 4
4.5 2.1

Sample Output:
14 6
6.6 2.4

[Solution Code]
#include <stdio.h>

int main() {
    int int1, int2;
    float float1, float2;

    if (scanf("%d %d", &int1, &int2) != 2) return 1;
    if (scanf("%f %f", &float1, &float2) != 2) return 1;

    printf("%d %d\n", int1 + int2, int1 - int2);
    printf("%.1f %.1f\n", float1 + float2, float1 - float2);

    return 0;
}

PROBLEM 4: Circle Area Calculation
--------------------------------------------------------------------------------
[Problem Description]
Calculates the area of a circle (Area = Pi * r^2) using an immutable single-precision 
floating-point constant for Pi (3.14159). Limits output to 2 decimal places.

Sample Input:
5.0

Sample Output:
78.54

[Solution Code]
#include <stdio.h>

int main() {
    const float PI = 3.14159f;
    float radius, area;

    if (scanf("%f", &radius) != 1) return 1;

    area = PI * radius * radius;
    printf("%.2f\n", area);

    return 0;
}

PROBLEM 5: Iterative Digital Root (Digit Reduction)
--------------------------------------------------------------------------------
[Problem Description]
Repeatedly adds all digits of an integer until a single digit remains. Supports
large boundary inputs up to INT_MAX (2,147,483,647) and zero.

Sample Input:
9875

Sample Output:
2

[Solution Code]
#include <stdio.h>

int main() {
    long long num;
    if (scanf("%lld", &num) != 1) return 1;

    if (num == 0) {
        printf("0\n");
        return 0;
    }

    long long result = (num % 9 == 0) ? 9 : (num % 9);
    printf("%lld\n", result);

    return 0;
}

PROBLEM 6: Temperature Extremes Tracking (Without Loops)
--------------------------------------------------------------------------------
[Problem Description]
Tracks the warmest and coldest temperatures from exactly 5 user-provided Celsius 
readings using basic variables without loops. The program asks for 5 sequential 
readings followed by a 999 completion signal. It initializes comparison trackers 
with INT_MIN and INT_MAX constants.

Sample Input 1:
-15 -28 -5 -32 -10

Sample Output 1:
Weather Station Report
Warmest Temperature: -5
Coldest Temperature: -32

Sample Input 2:
3 -8 12 0 -2 999

Sample Output 2:
Enter 5 temperatures followed by 999: --- Weather Station Report ---
Warmest Temperature: 12
Coldest Temperature: -8

[Solution Code]
#include <stdio.h>
#include <limits.h>

int main() {
    int t1, t2, t3, t4, t5, sentinel;
    int max = INT_MIN;
    int min = INT_MAX;

    // Read 5 readings and optional sentinel
    if (scanf("%d %d %d %d %d", &t1, &t2, &t3, &t4, &t5) != 5) return 1;
    scanf("%d", &sentinel); // Read 999 if present

    // Update max and min using basic conditional statements
    if (t1 > max) max = t1;
    if (t1 < min) min = t1;

    if (t2 > max) max = t2;
    if (t2 < min) min = t2;

    if (t3 > max) max = t3;
    if (t3 < min) min = t3;

    if (t4 > max) max = t4;
    if (t4 < min) min = t4;

    if (t5 > max) max = t5;
    if (t5 < min) min = t5;

    printf("Weather Station Report\n");
    printf("Warmest Temperature: %d\n", max);
    printf("Coldest Temperature: %d\n", min);

    return 0;
}

PROBLEM 7: Dynamic Unit Metric Converter Application
--------------------------------------------------------------------------------
[Problem Description]
Converts between Distance (KM/Miles) and Mass (KG/Pounds) based on user selection 
(1-4). Includes input sanitation filtering.

Option Matrix:
1: Kilometers to Miles (Factor: 0.621371)
2: Miles to Kilometers
3: Kilograms to Pounds (Factor: 2.20462)
4: Pounds to Kilograms

Sample Input:
1
100

Sample Output:
Result: 100.00 km = 62.14 miles

[Solution Code]
#include <stdio.h>

int main() {
    int choice;
    double rawValue, result;
    const double SCALE_FACTOR = 0.621371;
    const double WEIGHT_FACTOR = 2.20462;

    if (scanf("%d", &choice) != 1 || choice < 1 || choice > 4) {
        printf("[ERROR]: Invalid selection configuration. Terminating.\n");
        return 1;
    }

    if (scanf("%lf", &rawValue) != 1) return 1;

    switch(choice) {
        case 1:
            result = rawValue * SCALE_FACTOR;
            printf("Result: %.2lf km = %.2lf miles\n", rawValue, result);
            break;
        case 2:
            result = rawValue / SCALE_FACTOR;
            printf("Result: %.2lf miles = %.2lf km\n", rawValue, result);
            break;
        case 3:
            result = rawValue * WEIGHT_FACTOR;
            printf("Result: %.2lf kg = %.2lf lbs\n", rawValue, result);
            break;
        case 4:
            result = rawValue / WEIGHT_FACTOR;
            printf("Result: %.2lf lbs = %.2lf kg\n", rawValue, result);
            break;
    }

    return 0;
}
