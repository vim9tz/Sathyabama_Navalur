PROBLEM 7: Triangle Validator and Type
--------------------------------------------------------------------------------
Read three side lengths. First verify that every side is positive and that the
sum of every pair is greater than the third side. If valid, print Equilateral,
Isosceles, or Scalene. Otherwise print Invalid triangle.

Sample Input:  5 5 8
Sample Output: Isosceles triangle

[Solution Code]
#include <stdio.h>
int main(void) {
    int a, b, c;
    if (scanf("%d %d %d", &a, &b, &c) != 3) return 1;
    if (a <= 0 || b <= 0 || c <= 0 ||
        a + b <= c || a + c <= b || b + c <= a) {
        printf("Invalid triangle\n");
    } else if (a == b && b == c) {
        printf("Equilateral triangle\n");
    } else if (a == b || a == c || b == c) {
        printf("Isosceles triangle\n");
    } else {
        printf("Scalene triangle\n");
    }
    return 0;
}

PROBLEM 10: Electricity Bill
--------------------------------------------------------------------------------
Read the units consumed and calculate the bill using this progressive tariff:
    first 100 units       -> Rs. 1.50 per unit
    next 100 units        -> Rs. 2.50 per unit
    every unit above 200  -> Rs. 4.00 per unit
Reject negative units.

Sample Input:  250
Sample Output: Bill: Rs. 600.00

Explanation: (100 * 1.50) + (100 * 2.50) + (50 * 4.00) = 600.00

[Solution Code]
#include <stdio.h>
int main(void) {
    int units;
    double bill;
    if (scanf("%d", &units) != 1) return 1;
    if (units < 0) {
        printf("Invalid units\n");
        return 0;
    }
    if (units <= 100) bill = units * 1.50;
    else if (units <= 200) bill = 100 * 1.50 + (units - 100) * 2.50;
    else bill = 100 * 1.50 + 100 * 2.50 + (units - 200) * 4.00;
    printf("Bill: Rs. %.2f\n", bill);
    return 0;
}

PROBLEM 19: Sentinel-Controlled Temperature Tracker
--------------------------------------------------------------------------------
Repeatedly read temperatures until the user enters 999. Print the number of real
readings, warmest temperature, coldest temperature, and average. Do not include
999 in the calculations. If 999 is entered first, print No readings.

Sample Input:  -15 -28 -5 -32 -10 999
Sample Output:
Readings: 5
Warmest: -5
Coldest: -32
Average: -18.00

[Solution Code]
#include <limits.h>
#include <stdio.h>
int main(void) {
    int value, count = 0, warmest = INT_MIN, coldest = INT_MAX;
    long long sum = 0;
    while (scanf("%d", &value) == 1 && value != 999) {
        if (value > warmest) warmest = value;
        if (value < coldest) coldest = value;
        sum += value;
        count++;
    }
    if (count == 0) printf("No readings\n");
    else {
        printf("Readings: %d\n", count);
        printf("Warmest: %d\nColdest: %d\n", warmest, coldest);
        printf("Average: %.2f\n", (double)sum / count);
    }
    return 0;
}

PROBLEM 27: Prime Number Check
--------------------------------------------------------------------------------
Read an integer n. A prime number is greater than 1 and divisible only by 1 and
itself. Test possible divisors only while divisor * divisor <= n.

Sample Input:  29
Sample Output: Prime

[Solution Code]

int main(void) {
    int n, isPrime = 1;
    if (scanf("%d", &n) != 1) return 1;
    if (n <= 1) isPrime = 0;
    for (int divisor = 2; divisor <= n / divisor && isPrime; divisor++)
        if (n % divisor == 0) isPrime = 0;
    printf(isPrime ? "Prime\n" : "Not prime\n");
    return 0;
}


PROBLEM 28: All Prime Numbers in a Range
--------------------------------------------------------------------------------
Read lower and upper limits and print every prime number in that inclusive range.
This requires nested loops.

Sample Input:  10 25
Sample Output: 11 13 17 19 23

[Solution Code]
#include <stdio.h>
int main(void) {
    int lower, upper;
    if (scanf("%d %d", &lower, &upper) != 2 || lower > upper) return 1;
    for (int n = lower; n <= upper; n++) {
        int isPrime = n > 1;
        for (int divisor = 2; divisor <= n / divisor && isPrime; divisor++)
            if (n % divisor == 0) isPrime = 0;
        if (isPrime) printf("%d ", n);
    }
    printf("\n");
    return 0;
}

PROBLEM 33: Menu-Driven Unit Converter
--------------------------------------------------------------------------------
Read a menu choice and a value. Use `switch` to perform one conversion:

1. Kilometres to miles
2. Miles to kilometres
3. Kilograms to pounds
4. Pounds to kilograms

Sample Input:
1
100

Sample Output:
Result: 100.00 km = 62.14 miles

[Solution Code]
#include <stdio.h>

int main(void) {
    const double DISTANCE_FACTOR = 0.621371;
    const double WEIGHT_FACTOR = 2.20462;
    int choice;
    double value;

    if (scanf("%d %lf", &choice, &value) != 2) return 1;

    switch (choice) {
        case 1:
            printf("Result: %.2f km = %.2f miles\n",
                   value, value * DISTANCE_FACTOR);
            break;
        case 2:
            printf("Result: %.2f miles = %.2f km\n",
                   value, value / DISTANCE_FACTOR);
            break;
        case 3:
            printf("Result: %.2f kg = %.2f pounds\n",
                   value, value * WEIGHT_FACTOR);
            break;
        case 4:
            printf("Result: %.2f pounds = %.2f kg\n",
                   value, value / WEIGHT_FACTOR);
            break;
        default:
            printf("Invalid choice\n");
    }
    return 0;
}
